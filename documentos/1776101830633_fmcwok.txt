// §1 — FONTE (a causa raiz do texto invisível)
//
// @napi-rs/canvas no Linux NÃO usa fontes do sistema por padrão.
// A solução confirmada pelo repositório oficial:
//   1. Registrar via Buffer (não path)
//   2. Manter o buffer no escopo do módulo (evita garbage collection)
//   3. Usar aspas no nome da fonte no ctx.font
// ══════════════════════════════════════════════════════════════════════════════

let _fontBuf = null   // CRÍTICO: manter referência pra não ser GC'd
let FF = '"sans-serif"'  // fallback - pode não funcionar no Linux!

;(async function initFont() {
  const FONT_NAME = 'SP'  // SoulPet font

  // Tentativas em ordem de confiabilidade
  const cands = [
    // @fontsource/nunito (npm install @fontsource/nunito) — redondinha, fofa
    join(process.cwd(), 'node_modules/@fontsource/nunito/files/nunito-latin-700-normal.woff'),
    join(process.cwd(), 'node_modules/@fontsource/nunito/files/nunito-latin-400-normal.woff'),
    // @fontsource/roboto (fallback)
    join(process.cwd(), 'node_modules/@fontsource/roboto/files/roboto-latin-700-normal.woff'),
    join(process.cwd(), 'node_modules/@fontsource/roboto/files/roboto-latin-400-normal.woff'),
    // @fontsource/comic-neue (extra fofa)
    join(process.cwd(), 'node_modules/@fontsource/comic-neue/files/comic-neue-latin-700-normal.woff'),
    // Fontes do sistema Linux
    '/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf',
    '/usr/share/fonts/TTF/DejaVuSans-Bold.ttf',
    '/usr/share/fonts/truetype/liberation/LiberationSans-Bold.ttf',
    '/usr/share/fonts/truetype/ubuntu/Ubuntu-B.ttf',
    // Fonte cacheada de download anterior
    join(process.cwd(), 'data', 'sp_font.woff'),
  ]

  for (const p of cands) {
    if (!existsSync(p)) continue
    try {
      // Buffer.from() cria uma CÓPIA — evita problema de referência inválida
      _fontBuf = Buffer.from(readFileSync(p))
      GlobalFonts.register(_fontBuf, FONT_NAME)
      FF = `"${FONT_NAME}"`
      console.log(`[SoulPet] ✅ Fonte carregada: ${p}`)
      return
    } catch (e) { /* tenta próxima */ }
  }

  // Último recurso: baixar NotoSans do jsDelivr e cachear
  try {
    const { default: axios } = await import('axios')
    console.log('[SoulPet] 📥 Baixando fonte NotoSans...')
    const r = await axios.get(
      'https://cdn.jsdelivr.net/npm/@fontsource/noto-sans@5.0.0/files/noto-sans-latin-400-normal.woff',
      { responseType: 'arraybuffer', timeout: 20000 }
    )
    if (!existsSync(join(process.cwd(), 'data'))) mkdirSync(join(process.cwd(), 'data'), { recursive: true })
    const cachePath = join(process.cwd(), 'data', 'sp_font.woff')
    _fontBuf = Buffer.from(r.data)
    writeFileSync(cachePath, _fontBuf)
    GlobalFonts.register(_fontBuf, FONT_NAME)
    FF = `"${FONT_NAME}"`
    console.log('[SoulPet] ✅ Fonte baixada e cacheada!')
  } catch (e) {
    console.error('[SoulPet] ❌ Fonte não encontrada! Texto no card ficará invisível.')
    console.error('[SoulPet] ❌ Solução: npm install @fontsource/nunito')
    FF = '"sans-serif"'
  }
})()

// §4 — IA (Groq → Anthropic → fallback)
// ══════════════════════════════════════════════════════════════════════════════
async function iaCall(prompt, max=150){
  const cfg = getCfg()
  const gk  = cfg.groq_key || ''
  const ak  = process.env.ANTHROPIC_API_KEY || ''

  if (!gk && !ak) {
    console.log('[SP IA] Sem chave configurada — usando fallback')
    return null
  }

  if (gk) {
    try {
      const { default: axios } = await import('axios')
      console.log('[SP IA] → Groq llama-3.1-8b-instant...')
      const r = await axios.post(
        'https://api.groq.com/openai/v1/chat/completions',
        { model:'llama-3.1-8b-instant', max_tokens:max, temperature:0.9,
          messages:[{role:'user',content:prompt}] },
        { headers:{ Authorization:`Bearer ${gk}`, 'Content-Type':'application/json' }, timeout:12000 }
      )
      const txt = r.data.choices?.[0]?.message?.content?.trim()
      console.log('[SP IA] ✅ Groq OK:', txt?.slice(0,60))
      return txt || null
    } catch(e){
      console.error('[SP IA] ❌ Groq erro:', e.response?.data?.error?.message || e.message)
    }
  }

  if (ak) {
    try {
      const { default: axios } = await import('axios')
      console.log('[SP IA] → Anthropic...')
      const r = await axios.post(
        'https://api.anthropic.com/v1/messages',
        { model:'claude-haiku-4-5-20251001', max_tokens:max, messages:[{role:'user',content:prompt}] },
        { headers:{ 'x-api-key':ak, 'anthropic-version':'2023-06-01', 'Content-Type':'application/json' }, timeout:12000 }
      )
      const txt = r.data.content?.[0]?.text?.trim()
      console.log('[SP IA] ✅ Anthropic OK:', txt?.slice(0,60))
      return txt || null
    } catch(e){
      console.error('[SP IA] ❌ Anthropic erro:', e.response?.data || e.message)
    }
  }

  return null
}

async function iaDecideBatalha(pet, poderes, myHP, myMana, enHP, cds){
  const opts=['atacar','defender','curar']
  const pDisp=poderes.filter(p=>p.mana<=myMana&&!(cds[p.id]>0))
  pDisp.forEach(p=>opts.push(`poder ${p.id}`))
  if(myHP<25) opts.push('fugir')

  const r=await iaCall(
    `Você é ${pet.nome} em batalha.
HP:${~~myHP}/100 Mana:${~~myMana} | Inimigo HP:${~~enHP}
Opções: ${opts.join(' | ')}
CDs ativos: ${Object.entries(cds).filter(([,v])=>v>0).map(([k,v])=>`${k}(${v}t)`).join(',')||'nenhum'}
Escolha a MELHOR ação. Responda APENAS o comando exato, sem explicação.`,40)

  if(r){
    const c=r.toLowerCase().trim()
    for(const o of opts){ if(c.includes(o.replace('poder ','').split(' ')[0])) return o }
  }
  // Fallback inteligente
  if(myHP<30&&myMana>=12) return 'curar'
  if(pDisp.length&&myMana>=20&&Math.random()>.4) return `poder ${pDisp[~~(Math.random()*pDisp.length)].id}`
  return 'atacar'
}

const clamp  = (v,a=0,b=100) => Math.min(b,Math.max(a,v))
const calcH  = p => ~~(p.stats.felicidade*.4+p.stats.fome*.25+p.stats.sono*.2+p.stats.energia*.15)

function getHI(h){
  if(h>=80) return{emoji:'😄',label:'Radiante', hex:'#22c55e',bg:'#dcfce7'}
  if(h>=60) return{emoji:'😊',label:'Animado',  hex:'#84cc16',bg:'#ecfccb'}
  if(h>=40) return{emoji:'😐',label:'Neutro',   hex:'#eab308',bg:'#fefce8'}
  if(h>=25) return{emoji:'😒',label:'Irritado', hex:'#f97316',bg:'#fff7ed'}
  return      {emoji:'😠',label:'Furioso',  hex:'#ef4444',bg:'#fef2f2'}
}

// Sistema de Level
function calcNivel(totalXP){
  let lvl=0, acum=0
  while(true){ const needed=100+lvl*50; if(acum+needed>totalXP) break; acum+=needed; lvl++ }
  const needed=100+lvl*50, progresso=totalXP-acum
  return { lvl, xpAtual:progresso, xpProx:needed, pct:progresso/needed }
}

function ganharXP(pet, quantidade){
  pet.xpTotal = (pet.xpTotal||0)+quantidade
  const antes = calcNivel(pet.xpTotal-quantidade).lvl
  const depois = calcNivel(pet.xpTotal).lvl
  if(depois>antes) return depois  // level up!
  return null
}

function decay(pet){
  const now=Date.now(), h=(now-(pet.ultimaInteracao||now))/3_600_000
  if(h<.05) return pet
  const d=pet.dormindo, s=pet.stats
  s.fome       =clamp(s.fome      -CFG.DECAY.fome      *h)
  s.sono       =clamp(s.sono      +(d?4:-CFG.DECAY.sono)*h)
  s.felicidade =clamp(s.felicidade-CFG.DECAY.felicidade*h)
  s.energia    =clamp(s.energia   +(d?3:-CFG.DECAY.energia)*h)
  s.mana       =clamp(s.mana      +(d?2:-CFG.DECAY.mana)*h)
  s.humor      =calcH(pet)
  if(s.fome<=15) s.vida=clamp(s.vida-2*h)
  if(s.felicidade<10){ pet.felBaixa=pet.felBaixa||now; if((now-pet.felBaixa)>72*3_600_000&&pet.status==='vivo') pet.status='quebrado' }
  else pet.felBaixa=null
  if(s.vida<=10&&pet.status==='vivo') pet.status='fugiu'
  if(h>=1){ pet.histFel=pet.histFel||[]; pet.histFel.push(~~s.felicidade); if(pet.histFel.length>200) pet.histFel=pet.histFel.slice(-200) }
  pet.ultimaInteracao=now
  return pet
}

function getEstado(pet){
  const s=pet.stats
  if(pet.dormindo)               return 'dormindo'
  if(pet.status_temp==='doente') return 'doente'
  if(s.fome<20)                  return 'fome'
  if(s.humor<20)                 return 'bravo'
  if(s.felicidade<30)            return 'triste'
  if(s.energia<20)               return 'cansado'
  if(pet.evolucao?.forma==='Forma Lendária') return 'lendario'
  if(s.humor>=75)                return 'feliz'
  return 'padrao'
}

function hashStr(s){ let h=5381; for(const c of s) h=(h*33+c.charCodeAt(0))&0x7fffffff; return h }

function getPoderes(pet){
  const est=pet.evolucao?.estagio||0, tr=pet.tracos||[]
  const h=hashStr(pet.personalidade?.especie||pet.nome)
  const ids=new Set(['mordida'])

  // Segundo poder pelo hash da espécie (personalizado)
  const atks=['garra','chama','onda','trovao','sombra','pedrada']
  ids.add(atks[h%atks.length])

  // Traços desbloqueiam poderes específicos
  if(tr.includes('guerreiro'))   ids.add('inferno')
  if(tr.includes('protetor'))    { ids.add('escudo'); ids.add('espinhos') }
  if(tr.includes('carente'))     ids.add('cura_m')
  if(tr.includes('brincalhão'))  ids.add('brasa')
  if(tr.includes('disciplinado'))ids.add('roubo_m')
  if(tr.includes('glutão'))      ids.add('drenar')
  if(tr.includes('filho_eclipse'))ids.add('maldição')
  if(tr.includes('abençoado'))   { ids.add('sagrado'); ids.add('escudo_d') }

  // Estágio 2: especiais baseados no elemento da espécie
  if(est>=2){
    const spcs=['veneno','gelo','descarga','drenar','ilusão','ácido']
    ids.add(spcs[h%spcs.length])
    ids.add(spcs[(h+1)%spcs.length])
    ids.add('escudo')
    ids.add('cura_m')
  }

  // Estágio 3: poderes avançados + ultimate
  if(est>=3){
    ids.add('cura_M')
    const advs=['tempestade','tsunami','terremoto','sagrado']
    ids.add(advs[h%advs.length])
    // Ultimate baseado na forma
    const f=pet.evolucao?.forma||''
    if(f.includes('Lendária')||f.includes('Lendario')) { ids.add('apocalipse'); ids.add('renascer') }
    else if(f.includes('Sombra')) ids.add('devastação')
    else if(f.includes('Guardião')) ids.add('bênção')
    else ids.add('devastação')
  }

  return [...ids].map(id=>({id,...POOL[id]})).filter(p=>p.nome)
}

function getCharPath(uid, estado){
  for(const ext of ['png','jpg','jpeg','webp']){
    const p=join(CFG.CHARS,uid,`${estado}.${ext}`); if(existsSync(p)) return p
  }
  for(const ext of ['png','jpg','jpeg','webp']){
    const p=join(CFG.CHARS,uid,`padrao.${ext}`); if(existsSync(p)) return p
  }
  return null
}

// roundRect manual (sem bugs de versão)
function rr(ctx,x,y,w,h,r){ r=Math.min(r,w/2,h/2); ctx.beginPath(); ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r); ctx.arcTo(x+w,y+h,x,y+h,r); ctx.arcTo(x,y+h,x,y,r); ctx.arcTo(x,y,x+w,y,r); ctx.closePath() }

// Texto com outline (efeito cartoon)
function txtOutline(ctx, txt, x, y, col='#2d1b07', outlineCol='#fff', outlineW=3){
  ctx.strokeStyle=outlineCol; ctx.lineWidth=outlineW; ctx.lineJoin='round'; ctx.strokeText(txt,x,y)
  ctx.fillStyle=col; ctx.fillText(txt,x,y)
}

function checkMissaoProgress(pet, acao){
  if(!pet._missaoAtual||pet._missaoConcluida) return
  if(pet._missaoData!==new Date().toDateString()) return

    pet._missaoProgress=(pet._missaoProgress||0)+1
    if(pet._missaoProgress>=pet._missaoAtual.meta&&!pet._missaoNotif){
      pet._missaoNotif=true
    }
  }
}

function resolverAcao(uid,rawAcao,bat){
  const isP1=bat.p1===uid,me=isP1?1:2,en=isP1?2:1
  const [mhp,ehp,mmp,emp,msh,esh,mvn,evn,mstn,mcd]=[`hp${me}`,`hp${en}`,`mana${me}`,`mana${en}`,`shield${me}`,`shield${en}`,`veneno${me}`,`veneno${en}`,`stun${me}`,`cd${me}`]
  const logs=[]
  const acao=(rawAcao||'atacar').toLowerCase().trim()

  // Stun
  if(bat[mstn]>0){ bat[mstn]--; logs.push(`💫 ${acao} — atordoado! Perdeu o turno.`); bat.vez=en; bat.ultimoAtoTs=Date.now(); if(isP1) bat.cd1=tickCDs(bat.cd1); else bat.cd2=tickCDs(bat.cd2); return{logs,fim:false} }

  // Veneno passivo
  if(bat[mvn]>0){ bat[mhp]=Math.max(0,bat[mhp]-10); bat[mvn]--; logs.push(`☠️ Veneno: -10 vida (${bat[mvn]}t)`) }

  // Espinhos passivo (reflete dano)
  const espKey=`espinhos${me}`; let espinhosAtivo=bat[espKey]>0; if(espinhosAtivo) bat[espKey]--

  let acaoStr=''
  if(acao==='atacar'){
    const base=~~(Math.random()*18)+12, dano=Math.max(0,base-bat[esh]); bat[esh]=Math.max(0,bat[esh]-base); bat[ehp]=Math.max(0,bat[ehp]-dano)
    if(espinhosAtivo&&dano>0){ const ref=~~(dano*.5); bat[mhp]=Math.max(0,bat[mhp]-ref); acaoStr=`⚔️ -${dano} (espinhos devolveram ${ref}!)` }
    else acaoStr=`⚔️ Atacou! -${dano}${base-dano>0?` (escudo -${base-dano})`:''}` 
  } else if(acao==='defender'){
    bat[msh]+=40; bat[mmp]=Math.max(0,bat[mmp]-10); acaoStr=`🛡️ +40 escudo`
  } else if(acao==='curar'){
    const c=22; bat[mhp]=Math.min(100,bat[mhp]+c); bat[mmp]=Math.max(0,bat[mmp]-12); acaoStr=`💚 +${c} vida (HP: ${~~bat[mhp]})`
  } else if(acao.startsWith('poder ')){
    const pid=acao.slice(6).trim(), p=POOL[pid]
    if(!p){ const d=~~(Math.random()*12)+10; bat[ehp]=Math.max(0,bat[ehp]-d); acaoStr=`⚔️ Poder inválido → ataque básico: -${d}` }
    else if(bat[mmp]<(p.mana||0)){ const d=~~(Math.random()*10)+8; bat[ehp]=Math.max(0,bat[ehp]-d); acaoStr=`❌ Sem mana (${p.mana}) → ataque básico: -${d}` }
    else if((bat[mcd]||{})[pid]>0){ const d=~~(Math.random()*10)+8; bat[ehp]=Math.max(0,bat[ehp]-d); acaoStr=`⏳ ${p.nome} CD${bat[mcd][pid]}t → ataque: -${d}` }
    else{
      bat[mmp]-=p.mana||0; if(!bat[mcd]) bat[mcd]={}; bat[mcd][pid]=p.cd||0
      let fx=`${p.emoji} *${p.nome}*! `
      if(p.dano){ const d=Math.max(0,p.dano-bat[esh]); bat[esh]=Math.max(0,bat[esh]-p.dano); bat[ehp]=Math.max(0,bat[ehp]-d); fx+=`-${d} vida` }
      if(p.cura){ bat[mhp]=Math.min(100,bat[mhp]+p.cura); fx+=`+${p.cura} vida` }
      if(p.shield){ bat[msh]+=p.shield; fx+=`+${p.shield} escudo` }
      if(p.efeito==='veneno3'){ bat[evn]=3; fx+=` ☠️ envenenou!` }
      if(p.efeito==='veneno4'){ bat[evn]=4; fx+=` ☠️ veneno profundo!` }
      if(p.efeito==='roubo_mana'){ const r=Math.min(25,bat[emp]); bat[emp]-=r; bat[mmp]=Math.min(100,bat[mmp]+r); fx+=` 💜 +${r} mana` }
      if(p.efeito==='stun40'&&Math.random()<.4){ bat[`stun${en}`]=1; fx+=` 💫 inimigo atordoado!` }
      if(p.efeito==='freeze30'&&Math.random()<.3){ bat[`stun${en}`]=1; fx+=` ❄️ congelado!` }
      if(p.efeito==='drenar'){ const d=22; bat[ehp]=Math.max(0,bat[ehp]-d); bat[mhp]=Math.min(100,bat[mhp]+15); fx+=` drenando ${d}→+15` }
      if(p.efeito==='break_shield'){ bat[esh]=~~(bat[esh]*.5); fx+=` 💢 escudo cortado` }
      if(p.efeito==='corrode'){ bat[esh]=~~(bat[esh]*.5); fx+=` 🧪 escudo corroído` }
      if(p.efeito==='espinhos2'){ bat[`espinhos${me}`]=2; fx+=` 🌵 espinhos ativados!` }
      acaoStr=fx
    }
  } else if(acao==='fugir'){
    if(Math.random()<.35){ return{logs:[...logs,'🏃 Fugiu da batalha!'],fim:true,vencedor:en,fugiu:uid} }
    bat[mhp]=Math.max(0,bat[mhp]-12); acaoStr=`🏃 Tentou fugir mas falhou! -12 vida`
  } else {
    const d=~~(Math.random()*12)+10; bat[ehp]=Math.max(0,bat[ehp]-d); acaoStr=`⚔️ Ataque básico: -${d}`
  }

  logs.push(acaoStr)
  if(isP1) bat.cd1=tickCDs(bat.cd1||{}); else bat.cd2=tickCDs(bat.cd2||{})
  if(bat[ehp]<=0){ return{logs,fim:true,vencedor:me} }
  bat.vez=en; bat.ultimoAtoTs=Date.now(); bat.turno++
  bat.log=[...bat.log,...logs]; if(bat.log.length>30) bat.log=bat.log.slice(-30)
  const bs=getBatDB(); bs[bat.id]=bat; setBatDB(bs)
  return{logs,fim:false}
}

function checkGEv(uid,pet){

  const msgs=[]
  for(const stored of raw){
    if((stored.ap||[]).includes(uid)) continue
    const def=EVGLOB.find(x=>x.id===stored.id)
    if(!def) continue
    stored.ap=stored.ap||[]; stored.ap.push(uid)
    const d=def.fn(pet)

  }
  if(msgs.length){ setGEvDB(raw); setPet(uid,pet) }
  return msgs.join('')

function calcNegligencia(pet){
  const h=(Date.now()-(pet.ultimaInteracao||Date.now()))/3_600_000
  if(h<1) return 0

  if(h<12) return 2
  if(h<24) return 3
  return 4
}

function escolherComportamento(pet){
  const hora=getHoraLocal()
  const disponiveis=AUTO_COMPORTAMENTOS.filter(c=>{
    if(!c.trigger(pet,hora)) return false
    const ultimoTs=pet._autoHist?.[c.id]||0
    return Date.now()-ultimoTs>=c.intervalo(pet)
  })
  if(!disponiveis.length) return null
  const totalPeso=disponiveis.reduce((a,c)=>a+c.peso,0)
  let roll=Math.random()*totalPeso
  for(const c of disponiveis){ roll-=c.peso; if(roll<=0) return{c,hora} }
  return{c:disponiveis[0],hora}
}
