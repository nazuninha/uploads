--[=[
    P4 - Toogle mobile profissa
]=]

local UIS = game:GetService("UserInputService")
local Players = game:GetService("Players")
local lp = Players.LocalPlayer

if not UIS.TouchEnabled then return end

local gui = Instance.new("ScreenGui")
gui.Name = "MobileToggle"
gui.ResetOnSpawn = false
gui.Parent = lp:WaitForChild("PlayerGui")

local button = Instance.new("ImageButton")
button.Size = UDim2.new(0,80,0,80)
button.BackgroundTransparency = 1
button.Image = ToogleK7
button.ImageTransparency = 0
button.Parent = gui

local posFile = "mobile_toggle_pos.txt"
if isfile and isfile(posFile) then
    local p = string.split(readfile(posFile),",")
    local x = tonumber(p[1]) or 120
    local y = tonumber(p[2]) or 200
    button.Position = UDim2.new(0,x,0,y)
else
    button.Position = UDim2.new(0,120,0,200)
end

local isTouching = false
local moved = false
local dragStart = Vector2.new(0,0)
local startPos = UDim2.new()
local touchStartTime = 0
local hidden = false

local function savePos()
    if writefile then
        writefile(posFile, tostring(button.Position.X.Offset)..","..tostring(button.Position.Y.Offset))
    end
end

local function setHidden(state)
    hidden = state
    if hidden then
        button.ImageTransparency = 1
    else
        button.ImageTransparency = 0
    end
    savePos()
    WindUI:Notify({
        Title = "[ ! ] - Toggle",
        Content = hidden and "toogle ocultado" or "toogle mostrado",
        Duration = 3,
        Icon = subicon,
    })
end

local function longPressAction()
    setHidden(not hidden)
end

button.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch then
        isTouching = true
        moved = false
        dragStart = input.Position
        startPos = button.Position
        touchStartTime = tick()

        task.spawn(function()
            task.wait(2)
            if isTouching and not moved then
                longPressAction()
                isTouching = false
            end
        end)
    end
end)

button.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch and isTouching then
        local delta = input.Position - dragStart
        if not moved and delta.Magnitude > 8 then
            moved = true
        end
        if moved then
            button.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end
end)

button.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch then
        if isTouching then
            local duration = tick() - touchStartTime
            if not moved and duration < 2 then
                if hidden then
                    setHidden(false)
                    pcall(function() Window:Toggle() end)
                else
                    pcall(function() Window:Toggle() end)
                end
            else
                savePos()
            end
        end
        isTouching = false
        moved = false
    end
end)
Window:OnDestroy(function()
    local gui = game.Players.LocalPlayer.PlayerGui:FindFirstChild("MobileToggle")
if gui then
    gui:Destroy()
end
end)
