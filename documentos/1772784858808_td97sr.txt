local function try_request(opts)
    if request then
        local ok, res = pcall(request, opts)
        if not ok or not res then return nil end
        return res
    else
        local method = (opts.Method or "GET"):upper()
        if method == "GET" then
            local ok, body = pcall(function() return HttpService:GetAsync(opts.Url) end)
            if not ok then return nil end
            return { Body = body }
        elseif method == "POST" then
            local ok, body = pcall(function()
                return HttpService:PostAsync(opts.Url, opts.Body or "", Enum.HttpContentType.ApplicationJson)
            end)
            if not ok then return nil end
            return { Body = body }
        end
        return nil
    end
end

local function json_decode_safe(s)
    if not s then return nil end
    local ok, res = pcall(function() return HttpService:JSONDecode(s) end)
    if ok then return res end
    return nil
end

local function json_encode_safe(t)
    local ok, res = pcall(function() return HttpService:JSONEncode(t) end)
    if ok then return res end
    return "{}"
end

local function detect_hwid()
    local candidates = {
        _G.gethwid, _G.get_hwid, _G.GetHWID, _G.GetHwid,
        _G.getHWID, _G.identifyexecutor, _G.identify_executor,
        _G.GetExecutor, _G.GetHwId
    }
    for _,fn in ipairs(candidates) do
        if type(fn) == "function" then
            local ok, res = pcall(fn)
            if ok and res and res ~= "" then
                return tostring(res)
            end
        end
    end
    if syn and type(syn.get_endpoint) == "function" then
        local ok, res = pcall(function() return syn.get_endpoint() end)
        if ok and res then return tostring(res) end
    end
    return tostring(player.UserId or "unknown")
end

local function fetch_public_ip()
    local ok, res = pcall(function()
        return try_request({ Url = "https://api.ipify.org?format=json", Method = "GET" })
    end)
    if ok and res and res.Body then
        local dec = json_decode_safe(res.Body)
        if dec and dec.ip then return dec.ip end
    end
    return "unknown"
end

local function is_banned(bans, hwid, ip, nick)
    if type(bans) ~= "table" then return false, nil end
    for _,b in ipairs(bans) do
        if (b.hwid and hwid and tostring(b.hwid) == tostring(hwid))
        or (b.ip and ip and tostring(b.ip) == tostring(ip))
        or (b.account and nick and tostring(b.account) == tostring(nick)) then
            return true, b
        end
    end
    return false, nil
end

local RunService = game:GetService("RunService")
local player = game.Players.LocalPlayer

local speedEnabled = false
local speedValue = 50
local speedMode = "WalkSpeed [Tributo]"
local speedThread = nil
local bypassVelocity = nil
local lastRandomize = 0

local function applyDefaultSpeed()
    pcall(function()
        local char = player.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.WalkSpeed = 16
            hum.JumpPower = 50
        end
    end)
end

local function cleanupBypass()
    if bypassVelocity and bypassVelocity.Parent then
        bypassVelocity:Destroy()
    end
    bypassVelocity = nil
end

local function updateBypass(char, hum, root)
    if not (char and hum and root) then return end

    if not bypassVelocity or not bypassVelocity.Parent then
        cleanupBypass()
        bypassVelocity = Instance.new("BodyVelocity")
        bypassVelocity.Name = "BypassVelocity"
        bypassVelocity.MaxForce = Vector3.new(4000, 0, 4000)
        bypassVelocity.Velocity = Vector3.new(0, 0, 0)
        bypassVelocity.Parent = root
    end

    local moveDir = hum.MoveDirection
    if moveDir.Magnitude > 0 then
        bypassVelocity.Velocity = moveDir * speedValue
        root.CFrame = root.CFrame * CFrame.new(
            math.random(-1, 1) / 1000,
            0,
            math.random(-1, 1) / 1000
        )
    else
        bypassVelocity.Velocity = Vector3.new(0, 0, 0)
    end

    local now = tick()
    if now - lastRandomize > math.random(5, 15) / 100 then
        hum.WalkSpeed = math.random(14, 18)
        hum.JumpPower = math.random(45, 55)
        lastRandomize = now
    end
end

local function startSpeed()
    if speedEnabled then return end
    speedEnabled = true
    speedThread = task.spawn(function()
        while speedEnabled do
            pcall(function()
                local char = player.Character
                local hum = char and char:FindFirstChildOfClass("Humanoid")
                local root = char and char:FindFirstChild("HumanoidRootPart")
                if hum and root then
                    if speedMode == "WalkSpeed [Tributo]" then
                        hum.WalkSpeed = speedValue
                    elseif speedMode == "TpSpeed" then
                        hum.WalkSpeed = 0
                        if hum.MoveDirection and hum.MoveDirection.Magnitude > 0 then
                            root.CFrame = root.CFrame + (hum.MoveDirection * (speedValue / 20))
                        end
                    elseif speedMode == "Bypass [AntiDetect]" then
                        updateBypass(char, hum, root)
                    end
                end
            end)
            RunService.Heartbeat:Wait()
        end
        applyDefaultSpeed()
        cleanupBypass()
    end)
end

local function stopSpeed()
    speedEnabled = false
    applyDefaultSpeed()
    cleanupBypass()
end


if Window then
    Window:OnDestroy(function()
        stopSpeed()
    end)
end

local RunService = game:GetService("RunService")
local player = game.Players.LocalPlayer

local antiEnabled = false
local safePosition = Vector3.new(0, 0, 0) 
local antiThread = nil
local lastCFrame = nil

local function setSafePosition()
    pcall(function()
        local char = player.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if root then
            safePosition = root.Position
        end
    end)
end

local function startAntiTeleport()
    if antiEnabled then return end
    antiEnabled = true
    antiThread = task.spawn(function()
        while antiEnabled do
            pcall(function()
                local char = player.Character
                local root = char and char:FindFirstChild("HumanoidRootPart")
                if root then
                    local currentPos = root.Position
                    if lastCFrame then
                        local distance = (currentPos - lastCFrame.Position).Magnitude
                        if distance > 50 then -- teleporte suspeito
                            root.CFrame = CFrame.new(safePosition)
                        end
                    end
                    lastCFrame = root.CFrame
                end
            end)
            RunService.Heartbeat:Wait()
        end
        lastCFrame = nil
    end)
end

local function stopAntiTeleport()
    antiEnabled = false
    lastCFrame = nil
end
